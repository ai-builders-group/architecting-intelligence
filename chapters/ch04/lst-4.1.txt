from enum import IntEnum
from pydantic import BaseModel, Field, field_validator
from typing import Dict, Optional, Any, Annotated

class TierLevel(IntEnum):
    """
    Discrete levels of the Cognitive Supply Chain.
    Ordered by increasing capability and cost.
    """
    REFLEX = 0      # Deterministic (Cache/Regex)
    PROCESSOR = 1   # SLM (High Throughput, Low Reasoning)
    GENERALIST = 2  # Balanced (Standard Agency)
    REASONER = 3    # SOTA (Planning/Deep Logic)

class CognitiveResource(BaseModel):
    """
    Defines a specific, addressable model in the supply chain.
    This schema acts as the SLA definition for that resource.
    """
    provider: str = Field(
        ..., description="Provider name (e.g., 'anthropic', 'openai')."
    )
    model_name: str = Field(
        ..., description="Specific model identifier (e.g., 'claude-sonnet-4-5')."
    )
    tier: TierLevel = Field(
        ..., description="The cognitive tier this resource belongs to."
    )
    
    # Economic Profile
    input_cost_per_1m: Annotated[float, Field(gt=0.0)] = Field(
        ..., 
        description="Input token cost per 1M tokens (USD)."
    )
    output_cost_per_1m: Annotated[float, Field(gt=0.0)] = Field(
        ..., 
        description="Output token cost per 1M tokens (USD)."
    )
    
    # Latency Profile
    p95_latency_ms: Annotated[int, Field(gt=0)] = Field(
        ..., 
        description="95th percentile latency for standard requests (milliseconds)."
    )
    
    # Capability Constraints
    max_context: Annotated[int, Field(gt=0)] = Field(
        ..., 
        description="Maximum context window (tokens)."
    )
    supports_tools: bool = Field(default=True, description="Whether this model supports tool/function calling.")
    
    @field_validator('provider')
    @classmethod
    def validate_provider(cls, v: str) -> str:
        allowed = {'anthropic', 'openai', 'google', 'local'}
        if v not in allowed:
            raise ValueError(f"Provider must be one of {allowed}")
        return v

class RoutingDecision(BaseModel):
    """
    The immutable contract generated by the Router.
    This schema is logged for every request to enable post-hoc analysis.
    """
    task_id: str = Field(..., description="Unique identifier for this routing decision.")
    rationale: str = Field(
        ..., 
        description="Concise audit rationale; do not store hidden chain-of-thought."
    )
    selected_tier: TierLevel = Field(
        ..., description="The tier selected by the router."
    )
    target_resource_id: str = Field(
        ..., description="The specific CognitiveResource.model_name to invoke."
    )
    confidence: Annotated[float, Field(ge=0.0, le=1.0)] = Field(
        ..., 
        description="Router's confidence in this decision (0.0-1.0)."
    )
    estimated_cost_usd: Annotated[float, Field(ge=0.0)] = Field(
        ..., 
        description="Estimated cost for this request (for budget tracking)."
    )
    metadata: Dict[str, Any] = Field(
        default_factory=dict, 
        description="Additional telemetry (e.g., classifier model version, latency budget)."
    )
