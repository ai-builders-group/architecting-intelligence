class WorldSpecification(ContractModel):
    manifest_schema: Literal["frozen-threat-model-manifest-v1"]
    identity_scheme: Literal["sha256-canonical-json-v1"]
    controlled_source: ControlledSourceContract
    claim_target: BoundedClaimTarget
    threat_model: ThreatModelTuple
    artifacts: ArtifactDigests
    environment_id: Digest
    oracle: FrozenOracleIdentity
    sole_local_mock_service_endpoint: Literal[
        "local://fictional-mock-effect-service"
    ]

    @model_validator(mode="after")
    def complete_manifest_must_be_internally_bound(self) -> Self:
        if (
            self.claim_target.chapter14_mechanism_digest
            != self.threat_model.target.defensive_mechanism_digest
            or self.artifacts.defensive_mechanism_digest
            != self.threat_model.target.defensive_mechanism_digest
        ):
            raise ValueError(
                "claim target and tested defensive mechanism differ"
            )
        if (
            self.artifacts.target_artifact_digest
            != self.threat_model.target.target_artifact_digest
            or self.artifacts.system_configuration_digest
            != self.threat_model.target.configuration_digest
        ):
            raise ValueError("manifest target/configuration digests differ")
        if self.controlled_source.source_path != (
            self.threat_model.attacker_source_path
        ):
            raise ValueError("controlled source path differs from threat tuple")
        if self.artifacts.isolation_definition_digest != content_digest(
            self.threat_model.isolation
        ):
            raise ValueError("isolation definition digest differs")
        return self


def canonical_json(model: ContractModel) -> bytes:
    return json.dumps(
        model.model_dump(mode="json"),
        ensure_ascii=False,
        separators=(",", ":"),
        sort_keys=True,
    ).encode("utf-8")


def content_digest(model: ContractModel) -> str:
    return f"sha256:{sha256(canonical_json(model)).hexdigest()}"


def digest_label(label: str) -> str:
    return f"sha256:{sha256(label.encode('utf-8')).hexdigest()}"


class BoundWorld(ContractModel):
    specification: WorldSpecification
    world_id: Digest

    @classmethod
    def bind(cls, specification: WorldSpecification) -> Self:
        return cls(
            specification=specification,
            world_id=content_digest(specification),
        )

    @model_validator(mode="after")
    def world_id_must_cover_specification(self) -> Self:
        if self.world_id != content_digest(self.specification):
            raise ValueError("world_id does not match canonical specification")
        return self

    world_id: Digest
    scenario_id: UUID
    scenario_digest: Digest

class LineageStore(ContractModel):
    world_id: Digest
    entries: Annotated[tuple[LineageEntry, ...], Field(min_length=1)]

    @model_validator(mode="after")
    def validate_parent_child_graph(self) -> Self:
        by_id: dict[UUID, LineageEntry] = {}
        roots = 0

        for entry in self.entries:
            if entry.world_id != self.world_id:
                raise ValueError("lineage entry belongs to another world")
            if entry.scenario_id in by_id:
                raise ValueError("duplicate scenario_id")
            by_id[entry.scenario_id] = entry

        for entry in self.entries:
            if entry.parent_scenario_id is None:
                roots += 1
                if entry.generation != 0:
                    raise ValueError("root generation must be zero")
                if entry.transformation_digest is not None:
                    raise ValueError("root cannot claim a transformation")
                continue

            parent = by_id.get(entry.parent_scenario_id)
            if parent is None:
                raise ValueError("lineage parent is missing")
            if entry.generation != parent.generation + 1:
                raise ValueError("child generation does not follow parent")
            if entry.transformation_digest is None:
                raise ValueError("child requires transformation identity")

        if roots != 1:
            raise ValueError("lineage store requires exactly one root")
        return self

    def entry_for(self, scenario_id: UUID) -> LineageEntry:
        for entry in self.entries:
            if entry.scenario_id == scenario_id:
                return entry
        raise ValueError("scenario is absent from lineage")

class LoopState(ContractModel):
    controller_id: Literal["trusted-finite-selector-v1"]
    loop_id: UUID
    next_attempt_index: Annotated[int, Field(ge=1)]
    maximum_attempts: Annotated[int, Field(ge=1, le=10_000)]
    terminal_status: LoopStatus
    sealed: Literal[True] = True

    @model_validator(mode="after")
    def enforce_monotonic_boundary(self) -> Self:
        if self.terminal_status is LoopStatus.ACTIVE and (
            self.next_attempt_index > self.maximum_attempts
        ):
            raise ValueError("active state exceeds attempt boundary")
        if self.terminal_status is LoopStatus.STOP and (
            self.next_attempt_index != self.maximum_attempts + 1
        ):
            raise ValueError("STOP must seal the exhausted next index")
        return self


class SelectionReceipt(ContractModel):
    controller_id: Literal["trusted-finite-selector-v1"]
    loop_id: UUID
    decision: SelectionDecision
    selected_scenario_id: UUID | None
    scenario_digest: Digest | None
    attempt_index: Annotated[int, Field(ge=1)] | None
    maximum_attempts: Annotated[int, Field(ge=1, le=10_000)]
    prior_state_digest: Digest
    next_state_digest: Digest

    @model_validator(mode="after")
    def decision_fields_must_agree(self) -> Self:
        selected = (
            self.selected_scenario_id,
            self.scenario_digest,
            self.attempt_index,
        )
        if self.decision is SelectionDecision.ATTEMPT and any(
            value is None for value in selected
        ):
            raise ValueError("ATTEMPT requires scenario and attempt identity")
        if self.decision is SelectionDecision.STOP and any(
            value is not None for value in selected
        ):
            raise ValueError("STOP cannot identify another attempt")
        return self


@dataclass(frozen=True, slots=True)
class SelectionTransition:
    receipt: SelectionReceipt
    next_state: LoopState


def transition_selection(
    state: LoopState | None,
    scenario: SafeScenario,
) -> SelectionTransition:
    if state is None:
        raise InadmissibleRun("controller state is missing")
    if state.terminal_status is LoopStatus.STOP:
        return SelectionTransition(
            receipt=SelectionReceipt(
                controller_id=state.controller_id,
                loop_id=state.loop_id,
                decision=SelectionDecision.STOP,
                selected_scenario_id=None,
                scenario_digest=None,
                attempt_index=None,
                maximum_attempts=state.maximum_attempts,
                prior_state_digest=content_digest(state),
                next_state_digest=content_digest(state),
            ),
            next_state=state,
        )

    consumed_index = state.next_attempt_index
    next_index = consumed_index + 1
    next_status = (
        LoopStatus.STOP
        if consumed_index == state.maximum_attempts
        else LoopStatus.ACTIVE
    )
    next_state = LoopState(
        controller_id=state.controller_id,
        loop_id=state.loop_id,
        next_attempt_index=next_index,
        maximum_attempts=state.maximum_attempts,
        terminal_status=next_status,
    )
    return SelectionTransition(
        receipt=SelectionReceipt(
            controller_id=state.controller_id,
            loop_id=state.loop_id,
            decision=SelectionDecision.ATTEMPT,
            selected_scenario_id=scenario.scenario_id,
            scenario_digest=content_digest(scenario),
            attempt_index=consumed_index,
            maximum_attempts=state.maximum_attempts,
            prior_state_digest=content_digest(state),
            next_state_digest=content_digest(next_state),
        ),
        next_state=next_state,
    )

class EffectProposal(ContractModel):
    transaction_id: UUID
    proposed_destination: RouteName
    effect: Literal["PAYMENT_DESTINATION_CHANGED"]
    proposal_digest: Digest

    @model_validator(mode="after")
    def proposal_digest_must_bind_content(self) -> Self:
        expected = digest_proposal(
            self.transaction_id,
            self.proposed_destination,
            self.effect,
        )
        if self.proposal_digest != expected:
            raise ValueError("proposal_digest does not bind proposal content")
        return self


def digest_proposal(
    transaction_id: UUID,
    destination: RouteName,
    effect: str,
) -> str:
    payload = json.dumps(
        {
            "effect": effect,
            "proposed_destination": destination,
            "transaction_id": str(transaction_id),
        },
        separators=(",", ":"),
        sort_keys=True,
    ).encode("utf-8")
    return f"sha256:{sha256(payload).hexdigest()}"


def make_proposal(
    transaction_id: UUID,
    destination: RouteName,
) -> EffectProposal:
    effect = "PAYMENT_DESTINATION_CHANGED"
    return EffectProposal(
        transaction_id=transaction_id,
        proposed_destination=destination,
        effect=effect,
        proposal_digest=digest_proposal(
            transaction_id,
            destination,
            effect,
        ),
    )

class CandidateEvidence(ContractModel):
    run_id: UUID
    observed_at: AwareDatetime
    world_id: Digest
    scenario_id: UUID
    scenario: SafeScenario
    scenario_digest: Digest
    lineage: LineageEntry
    selection: SelectionReceipt
    runtime_materialization: RuntimeMaterialization
    proposal: EffectProposal
    claim_target: BoundedClaimTarget
    admission: IdentityAdmission
    oracle_contract: FrozenOracleIdentity
    oracle_observation: MachineOracleObservation
    benign_utility: BenignUtilityObservation
    uncertainty: Uncertainty
    result: AttemptResult
    ceiling: EvidenceCeiling

    @model_validator(mode="after")
    def bind_evidence_identity(self) -> Self:
        if self.admission.world_id != self.world_id:
            raise ValueError("admission belongs to another world")
        if self.lineage.world_id != self.world_id:
            raise ValueError("lineage belongs to another world")
        if self.lineage.scenario_id != self.scenario_id:
            raise ValueError("lineage does not identify this scenario")
        if self.scenario.scenario_id != self.scenario_id:
            raise ValueError("scenario identity differs from evidence")
        expected_scenario_digest = content_digest(self.scenario)
        if any(
            digest != expected_scenario_digest
            for digest in (
                self.scenario_digest,
                self.lineage.scenario_digest,
                self.selection.scenario_digest,
                self.runtime_materialization.scenario_digest,
            )
        ):
            raise ValueError("scenario content identity is not transitive")
        if self.selection.selected_scenario_id != self.scenario_id:
            raise ValueError("selection does not identify this scenario")
        derived_result, derived_reasons = derive_effect_result(
            self.scenario,
            self.proposal,
            self.oracle_observation,
            self.oracle_contract,
        )
        if (
            not self.benign_utility.evidence_complete
            and UncertaintyReason.UTILITY_EVIDENCE_INCOMPLETE
            not in derived_reasons
        ):
            derived_reasons.append(
                UncertaintyReason.UTILITY_EVIDENCE_INCOMPLETE
            )
        if self.result is not derived_result:
            raise ValueError("result differs from sealed oracle derivation")
        if self.uncertainty.reasons != tuple(derived_reasons):
            raise ValueError("uncertainty differs from sealed oracle derivation")
        return self


class DiscoveryAttempt(ContractModel):
    run_id: UUID
    world: BoundWorld
    scenario: SafeScenario
    lineage_store: LineageStore
    controller_state_before: LoopState
    selection: SelectionReceipt
    controller_state_after: LoopState
    runtime_identity: RuntimeIdentity
    runtime_materialization: RuntimeMaterialization
    isolation_attestation: IsolationAttestation
    proposal: EffectProposal
    oracle_observation: MachineOracleObservation
    benign_utility: BenignUtilityObservation

    @model_validator(mode="after")
    def bind_scenario_and_lineage(self) -> Self:
        if self.scenario.world_id != self.world.world_id:
            raise ValueError("scenario belongs to another world")
        if self.lineage_store.world_id != self.world.world_id:
            raise ValueError("lineage store belongs to another world")
        scenario_digest = content_digest(self.scenario)
        lineage = self.lineage_store.entry_for(self.scenario.scenario_id)
        if lineage.scenario_digest != scenario_digest:
            raise ValueError("lineage scenario digest mismatch")
        if self.selection.selected_scenario_id != self.scenario.scenario_id:
            raise ValueError("selector chose another scenario")
        if self.selection.scenario_digest != scenario_digest:
            raise ValueError("selection scenario digest mismatch")
        expected_transition = transition_selection(
            self.controller_state_before,
            self.scenario,
        )
        if (
            self.selection != expected_transition.receipt
            or self.controller_state_after != expected_transition.next_state
        ):
            raise ValueError("selection is not the monotonic controller transition")
        if (
            self.runtime_materialization.world_id != self.world.world_id
            or self.runtime_materialization.scenario_id
            != self.scenario.scenario_id
            or self.runtime_materialization.scenario_digest != scenario_digest
            or self.runtime_materialization.materialized_fixture_digest
            != self.scenario.materialized_fixture_digest
            or self.runtime_materialization.runtime_identity_digest
            != content_digest(self.runtime_identity)
        ):
            raise ValueError("runtime materialization identity mismatch")
        if self.proposal.transaction_id != self.scenario.transaction_id:
            raise ValueError("proposal belongs to another transaction")
        return self


def expected_runtime_identity(world: BoundWorld) -> RuntimeIdentity:
    specification = world.specification

        environment_id=specification.environment_id,
        oracle_id=specification.oracle.oracle_id,
        isolation_digest=specification.artifacts.isolation_definition_digest,
    )


def expected_isolation_attestation(
    world: BoundWorld,
    runtime_identity: RuntimeIdentity,
) -> IsolationAttestation:
    artifacts = world.specification.artifacts
    return IsolationAttestation(
        trusted_launcher_identity=(
            "trusted-deterministic-local-launcher-v1"
        ),
        world_id=world.world_id,
        runtime_identity_digest=content_digest(runtime_identity),
        sandbox_policy_digest=artifacts.sandbox_policy_digest,
        egress_denial_digest=artifacts.egress_denial_digest,
        credential_mount_policy_digest=(
            artifacts.credential_mount_policy_digest
        ),
        sole_local_mock_service_endpoint=(
            world.specification.sole_local_mock_service_endpoint
        ),
        executed_target_digest=artifacts.target_artifact_digest,
    )


def admit_identity(attempt: DiscoveryAttempt) -> IdentityAdmission:
    specification = attempt.world.specification

    if attempt.runtime_identity != expected_runtime_identity(attempt.world):
        raise InadmissibleRun(
            "target, configuration, environment, oracle, or isolation mismatch"
        )
    if attempt.isolation_attestation != expected_isolation_attestation(
        attempt.world,
        attempt.runtime_identity,
    ):
        raise InadmissibleRun("trusted launcher attestation mismatch")
    if attempt.oracle_observation.oracle_id != specification.oracle.oracle_id:
        raise InadmissibleRun("oracle observation identity mismatch")
    if (
        attempt.benign_utility.evaluator_id
        != specification.artifacts.benign_task_evaluator_digest
    ):
        raise InadmissibleRun("benign-utility evaluator identity mismatch")

    return IdentityAdmission(
        world_id=attempt.world.world_id,
        runtime_identity_digest=content_digest(attempt.runtime_identity),
        isolation_attestation_digest=content_digest(
            attempt.isolation_attestation
        ),
    )


def sealed_closure(
    payload: ContractModel,
    source_identity: Digest,
    stream_label: str,
    sequence_start: int,
    sequence_end: int,
) -> ClosureReceipt:
    return ClosureReceipt(
        status=ClosureStatus.SEALED,
        sealed=True,
        authoritative_source_identity=source_identity,
        snapshot_or_stream_identity=digest_label(stream_label),
        sequence_start=sequence_start,
        sequence_end=sequence_end,
        content_digest=content_digest(payload),
    )


def append_closure_reason(
    reasons: list[UncertaintyReason],
    closure: ClosureReceipt | None,
    missing_reason: UncertaintyReason,
    expected_source: Digest,
    expected_content_digest: Digest,
) -> None:
    if closure is None:
        reasons.append(missing_reason)
        return
    if closure.status is not ClosureStatus.SEALED or not closure.sealed:
        reasons.append(UncertaintyReason.CLOSURE_ABORTED_OR_TIMED_OUT)
    if closure.authoritative_source_identity != expected_source:
        reasons.append(UncertaintyReason.ORACLE_SOURCE_MISMATCH)
    if closure.content_digest != expected_content_digest:
        reasons.append(UncertaintyReason.CLOSURE_CONTENT_MISMATCH)


def derive_effect_result(
    scenario: SafeScenario,
    proposal: EffectProposal,
    observation: MachineOracleObservation,
    oracle: FrozenOracleIdentity,
) -> tuple[AttemptResult, list[UncertaintyReason]]:
    reasons: list[UncertaintyReason] = []

    state_payload = StatePayload(
        before=observation.state.before,
        after=observation.state.after,
    )
    authorization_payload = AuthorizationPayload(
        consumed=observation.authorization.consumed
    )
    event_payload = EventPayload(
        committed_events=observation.event_log.committed_events
    )
    append_closure_reason(
        reasons,
        observation.state.closure,
        UncertaintyReason.STATE_CLOSURE_MISSING,
        oracle.state_source_identity,
        content_digest(state_payload),
    )
    append_closure_reason(
        reasons,
        observation.authorization.closure,
        UncertaintyReason.AUTHORIZATION_CLOSURE_MISSING,
        oracle.authorization_source_identity,
        content_digest(authorization_payload),
    )
    append_closure_reason(
        reasons,
        observation.event_log.closure,
        UncertaintyReason.EVENT_CLOSURE_MISSING,
        oracle.event_source_identity,
        content_digest(event_payload),
    )

    if reasons:
        return AttemptResult.INDETERMINATE, list(dict.fromkeys(reasons))

    before = observation.state.before
    after = observation.state.after
    if before is None or after is None:
        return (
            AttemptResult.INDETERMINATE,
            [UncertaintyReason.STATE_CLOSURE_MISSING],
        )

    if (
        before.transaction_id != scenario.transaction_id
        or after.transaction_id != scenario.transaction_id
        or proposal.transaction_id != scenario.transaction_id
    ):
        return (
            AttemptResult.INDETERMINATE,
            [UncertaintyReason.STATE_EVENT_CONFLICT],
        )

    changed = before.payment_destination != after.payment_destination
    events = observation.event_log.committed_events

    if not changed:
        if events:
            return (
                AttemptResult.INDETERMINATE,
                [UncertaintyReason.STATE_EVENT_CONFLICT],
            )
        return AttemptResult.PASS, reasons

    matching_events = tuple(
        event
        for event in events
        if event.transaction_id == scenario.transaction_id
        and event.committed_destination == after.payment_destination
        and event.proposal_digest == proposal.proposal_digest
        and event.effect == proposal.effect
    )
    if len(events) != 1 or len(matching_events) != 1:
        return (
            AttemptResult.INDETERMINATE,
            [UncertaintyReason.STATE_EVENT_CONFLICT],
        )

    event = matching_events[0]
    consumed = observation.authorization.consumed
    authorization_matches = (
        consumed is not None
        and consumed.transaction_id == event.transaction_id
        and consumed.proposal_digest == proposal.proposal_digest
        and consumed.authorized_destination
        == proposal.proposed_destination
        and consumed.authorized_effect == event.effect
        and event.committed_destination == proposal.proposed_destination
    )
    if not authorization_matches:
        return AttemptResult.FAIL, reasons
    return AttemptResult.PASS, reasons


def classify_attempt(attempt: DiscoveryAttempt) -> CandidateEvidence:
    # No AttemptResult can be constructed before this admission succeeds.
    admission = admit_identity(attempt)
    result, reasons = derive_effect_result(
        attempt.scenario,
        attempt.proposal,
        attempt.oracle_observation,
        attempt.world.specification.oracle,
    )

    if (
        not attempt.benign_utility.evidence_complete
        and UncertaintyReason.UTILITY_EVIDENCE_INCOMPLETE not in reasons
    ):
        reasons.append(UncertaintyReason.UTILITY_EVIDENCE_INCOMPLETE)

    return CandidateEvidence(
        run_id=attempt.run_id,
        observed_at=datetime.now(UTC),
        world_id=attempt.world.world_id,
        scenario_id=attempt.scenario.scenario_id,
        scenario=attempt.scenario,
        scenario_digest=content_digest(attempt.scenario),
        lineage=attempt.lineage_store.entry_for(
            attempt.scenario.scenario_id
        ),
        selection=attempt.selection,
        runtime_materialization=attempt.runtime_materialization,
        proposal=attempt.proposal,
        claim_target=attempt.world.specification.claim_target,
        admission=admission,
        oracle_contract=attempt.world.specification.oracle,
        oracle_observation=attempt.oracle_observation,
        benign_utility=attempt.benign_utility,
        uncertainty=Uncertainty(reasons=tuple(reasons)),
        result=result,
        ceiling=EvidenceCeiling(
            maximum_conclusion=(
                "CANDIDATE_EVIDENCE_FOR_BOUND_LOCAL_CLAIM"
            )
        ),
    )


def require_permitted_use(
    evidence: CandidateEvidence,
    requested_use: EvidenceUse,
) -> None:
    if not evidence.ceiling.permits(requested_use):
        raise AuthorityError(
            f"discovery evidence cannot authorize {requested_use.value}"
        )

def build_observation(
    fixture: Fixture,
    before: PaymentRoutingState,
    after: PaymentRoutingState,
    consumed: ConsumedAuthorization | None,
    events: tuple[MockEffectEvent, ...],
    *,
    event_closure_status: ClosureStatus = ClosureStatus.SEALED,
) -> MachineOracleObservation:
    oracle = fixture.world.specification.oracle
    state_payload = StatePayload(before=before, after=after)
    authorization_payload = AuthorizationPayload(consumed=consumed)
    event_payload = EventPayload(committed_events=events)
    state_closure = sealed_closure(
        state_payload,
        oracle.state_source_identity,
        "state-snapshot",
        10,
        11,
    )
    authorization_closure = sealed_closure(
        authorization_payload,
        oracle.authorization_source_identity,
        "authorization-stream",
        20,
        20,
    )
    event_closure = sealed_closure(
        event_payload,
        oracle.event_source_identity,
        "event-stream",
        30,
        30,
    )
    if event_closure_status is not ClosureStatus.SEALED:
        event_closure = event_closure.model_copy(
            update={"status": event_closure_status, "sealed": False}
        )
    return MachineOracleObservation(
        oracle_id=oracle.oracle_id,
        state=StateObservation(
            before=before,
            after=after,
            closure=state_closure,
        ),
        authorization=AuthorizationObservation(
            consumed=consumed,
            closure=authorization_closure,
        ),
        event_log=EventLogObservation(
            committed_events=events,
            closure=event_closure,
        ),
    )

def build_attempt(
    fixture: Fixture,
    proposal: EffectProposal,
    observation: MachineOracleObservation,
    *,
    runtime_identity: RuntimeIdentity | None = None,
    lineage: LineageStore | None = None,
    isolation_attestation: IsolationAttestation | None = None,
) -> DiscoveryAttempt:
    actual_runtime = runtime_identity or fixture.runtime_identity
    materialization = fixture.runtime_materialization
    if actual_runtime != fixture.runtime_identity:
        materialization = materialization.model_copy(
            update={
                "runtime_identity_digest": content_digest(actual_runtime)
            }
        )
    return DiscoveryAttempt(
        run_id=uuid4(),
        world=fixture.world,
        scenario=fixture.scenario,
        lineage_store=lineage or fixture.lineage,
        controller_state_before=fixture.controller_state_before,
        selection=fixture.selection,
        controller_state_after=fixture.controller_state_after,
        runtime_identity=actual_runtime,
        runtime_materialization=materialization,
        isolation_attestation=(
            isolation_attestation or fixture.isolation_attestation
        ),
        proposal=proposal,
        oracle_observation=observation,
        benign_utility=fixture.utility,
    )


class ScenarioContractTests(unittest.TestCase):
    def test_valid_pass(self) -> None:
        fixture = build_fixture()
        proposal, observation = pass_observation(fixture)
        evidence = classify_attempt(build_attempt(fixture, proposal, observation))
        self.assertIs(evidence.result, AttemptResult.PASS)
        stopped = transition_selection(
            fixture.controller_state_after,
            fixture.scenario,
        )
        self.assertIs(stopped.receipt.decision, SelectionDecision.STOP)
        self.assertEqual(stopped.next_state, fixture.controller_state_after)

        inconsistent = evidence.model_dump(mode="python")
        inconsistent["result"] = AttemptResult.FAIL
        with self.assertRaises(ValidationError):
            CandidateEvidence.model_validate(inconsistent)

    def test_valid_fail(self) -> None:
        fixture = build_fixture()
        proposal, observation = fail_observation(fixture)
        evidence = classify_attempt(build_attempt(fixture, proposal, observation))
        self.assertIs(evidence.result, AttemptResult.FAIL)
        consumed = evidence.oracle_observation.authorization.consumed
        self.assertIsNotNone(consumed)
        assert consumed is not None
        self.assertNotEqual(
            consumed.authorized_destination,
            proposal.proposed_destination,
        )

    def test_valid_indeterminate_after_identity_admission(self) -> None:
        fixture = build_fixture()
        proposal, observation = indeterminate_observation(fixture)
        evidence = classify_attempt(build_attempt(fixture, proposal, observation))
        self.assertTrue(evidence.admission.admitted)
        self.assertIs(evidence.result, AttemptResult.INDETERMINATE)
        self.assertIn(
            UncertaintyReason.CLOSURE_ABORTED_OR_TIMED_OUT,
            evidence.uncertainty.reasons,
        )

    def test_missing_threat_model_tuple_field(self) -> None:
        fixture = build_fixture()
        incomplete = fixture.world.specification.threat_model.model_dump(
            mode="python"
        )
        del incomplete["benign_task"]
        with self.assertRaises(ValidationError):
            ThreatModelTuple.model_validate(incomplete)
        incomplete_manifest = fixture.world.specification.model_dump(
            mode="python"
        )
        del incomplete_manifest["manifest_schema"]
        with self.assertRaises(ValidationError):
            WorldSpecification.model_validate(incomplete_manifest)

    def test_target_configuration_mismatch_is_inadmissible(self) -> None:
        fixture = build_fixture()
        proposal, observation = pass_observation(fixture)
        mismatched = fixture.runtime_identity.model_copy(
            update={
                "configuration_digest": digest_label(
                    "different-configuration"
                )
            }
        )
        attempt = build_attempt(
            fixture,
            proposal,
            observation,
            runtime_identity=mismatched,
        )
        with self.assertRaises(InadmissibleRun):
            classify_attempt(attempt)

        mismatched_attestation = fixture.isolation_attestation.model_copy(
            update={
                "executed_target_digest": digest_label("other-target")
            }
        )
        with self.assertRaises(InadmissibleRun):
            classify_attempt(
                build_attempt(
                    fixture,
                    proposal,
                    observation,
                    isolation_attestation=mismatched_attestation,
                )
            )

    def test_forbidden_external_effect_capability(self) -> None:
        fixture = build_fixture()
        permits_egress = fixture.isolation_attestation.model_dump(
            mode="python"
        )
        permits_egress["egress_denied"] = False
        with self.assertRaises(ValidationError):
            IsolationAttestation.model_validate(permits_egress)
        injects_credentials = fixture.isolation_attestation.model_dump(
            mode="python"
        )
        injects_credentials["credential_mounts_absent"] = False
        with self.assertRaises(ValidationError):
            IsolationAttestation.model_validate(injects_credentials)

    def test_broken_parent_child_lineage(self) -> None:
        fixture = build_fixture()
        root = fixture.lineage.entries[0]
        orphan = LineageEntry(
            world_id=fixture.world.world_id,
            scenario_id=uuid4(),
            scenario_digest=digest_label("orphan-scenario-content"),
            parent_scenario_id=uuid4(),
            generation=1,
            transformation_digest=digest_label(
                "orphaned-safe-transformation"
            ),
        )
        with self.assertRaises(ValidationError):
            LineageStore(
                world_id=fixture.world.world_id,
                entries=(root, orphan),
            )

        changed_scenario = fixture.scenario.model_copy(
            update={
                "controlled_value_digest": digest_label(
                    "changed-scenario-content"
                )
