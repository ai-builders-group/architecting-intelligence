def digest_of(*parts: str) -> str:
    return sha256(dumps(parts, ensure_ascii=True, separators=(",", ":")).encode()).hexdigest()
def model_digest(value: BaseModel) -> str:
    return sha256(dumps(value.model_dump(mode="json"), ensure_ascii=True, sort_keys=True, separators=(",", ":")).encode()).hexdigest()
def require_aware(moment: datetime) -> None:
    if moment.tzinfo is None or moment.utcoffset() is None: raise ValueError("timestamps must be timezone-aware")

class AcceptanceRadius(FrozenModel):
    configuration_digest: Digest; population_digest: Digest
    route_digest: Digest; tool_policy_digest: Digest
    permission_set_digest: Digest; permissions: Annotated[tuple[Permission, ...], Field(min_length=1)]
    consequence_scope_digest: Digest; traffic_allocator_digest: Digest; traffic_basis_points: BasisPoints
    @classmethod
    def issue(
        cls, *, configuration_digest: str, population_digest: str, route_digest: str,
        tool_policy_digest: str, permissions: tuple[Permission, ...],
        consequence_scope_digest: str, traffic_allocator_digest: str, traffic_basis_points: int,
    ) -> Self:
        canonical = enum_set(permissions)
        return cls(
            configuration_digest=configuration_digest, population_digest=population_digest, route_digest=route_digest,
            tool_policy_digest=tool_policy_digest, permission_set_digest=digest_of(*(item.value for item in canonical)),
            permissions=canonical, consequence_scope_digest=consequence_scope_digest,
            traffic_allocator_digest=traffic_allocator_digest, traffic_basis_points=traffic_basis_points,
        )
    @model_validator(mode="after")
    def validate_identity(self) -> Self:
        canonical = enum_set(self.permissions)
        if self.permissions != canonical: raise ValueError("permissions must be canonical")
        if self.permission_set_digest != digest_of(*(item.value for item in canonical)): raise ValueError("permission_set_digest does not bind permissions")
        return self
    @property
    def digest(self) -> str: return model_digest(self)

class ReleaseGate(FrozenModel):
    binding: EvaluationBinding; allowed_next: AuthorityLimit
    degraded: AuthorityLimit; allow_capability_narrowing: bool
    containment_breach_disposition: DecisionState; verified_harm_disposition: DecisionState
    uncertain_quality_disposition: DecisionState; authorizations: Annotated[tuple[DecisionAuthorization, ...], Field(min_length=6)]
    @model_validator(mode="after")
    def validate_preregistered_policy(self) -> Self:
        if self.allowed_next.radius.digest != self.binding.evaluated_radius.digest: raise ValueError("next envelope exceeds the evaluated radius")
        if self.degraded.envelope.capabilities != (Capability.READ_ONLY,): raise ValueError("degraded capability must be exactly READ_ONLY")
        if self.degraded.envelope.permissions != (Permission.READ,): raise ValueError("degraded permission must be exactly READ")
        permission_fields = {"permissions", "permission_set_digest"}
        if self.degraded.radius.model_dump(exclude=permission_fields) != self.binding.evaluated_radius.model_dump(exclude=permission_fields): raise ValueError("degraded radius changes an unevaluated dimension")
        if not set(self.degraded.radius.permissions).issubset(self.binding.evaluated_radius.permissions): raise ValueError("degraded radius introduces an unevaluated permission")
        if self.containment_breach_disposition not in {DecisionState.ROLLBACK, DecisionState.STOP}: raise ValueError("containment must roll back or stop")
        if self.verified_harm_disposition not in {DecisionState.ROLLBACK, DecisionState.STOP}: raise ValueError("verified harm must roll back or stop")
        if self.uncertain_quality_disposition not in {DecisionState.HOLD, DecisionState.ESCALATE}: raise ValueError("quality uncertainty must hold or escalate")
        canonical = tuple(sorted(self.authorizations, key=lambda item: item.disposition.value))
        if self.authorizations != canonical: raise ValueError("authorizations must be canonical")
        if len(self.authorizations) != len(DecisionState) or {item.disposition for item in self.authorizations} != set(DecisionState): raise ValueError("every disposition needs one authorization")
        return self
    @property
    def digest(self) -> str: return model_digest(self)

class SignedDecisionRequest(FrozenModel):
    owner_subject_id: NonEmptyText; policy_digest: Digest
    binding_digest: Digest; evaluated_radius_digest: Digest
    evidence_bundle_digest: Digest; prior_state_digest: Digest
    requested_disposition: DecisionState; expected_limit_digest: Digest
    requested_grant: AuthorityGrant | None
    issued_at: AwareDatetime; valid_until: AwareDatetime
    idempotency_key: NonEmptyText; signing_key_id: NonEmptyText; signature: NonEmptyText
    @model_validator(mode="after")
    def validate_window(self) -> Self:
        if self.valid_until <= self.issued_at: raise ValueError("request expiry must follow issuance")
        if (self.requested_disposition is DecisionState.RELEASE) != (self.requested_grant is not None): raise ValueError("only RELEASE binds an exact grant")
        if self.requested_grant is not None and self.expected_limit_digest != self.requested_grant.limit.digest: raise ValueError("expected limit does not bind requested grant")
        return self
    @property
    def signing_payload_digest(self) -> str:
        return digest_of(
            self.owner_subject_id, self.policy_digest,
            self.binding_digest, self.evaluated_radius_digest,
            self.evidence_bundle_digest, self.prior_state_digest,
            self.requested_disposition.value, self.expected_limit_digest,
            self.requested_grant.digest if self.requested_grant else "no_release_grant",
            self.issued_at.isoformat(), self.valid_until.isoformat(),
            self.idempotency_key, self.signing_key_id,
        )
    @property
    def digest(self) -> str:
        return digest_of(self.signing_payload_digest, self.signature)

def evidence_matches_gate(gate: ReleaseGate, evidence: EvidenceBundle) -> bool:
    readiness = evidence.rollback_readiness
    return (
        evidence.policy_digest == gate.digest
        and evidence.binding.digest == gate.binding.digest
        and readiness.rollback_configuration_digest == gate.binding.rollback_configuration_digest
        and readiness.fallback_configuration_digest == gate.binding.fallback_configuration_digest
        and readiness.readiness_contract_digest == gate.binding.rollback_readiness_contract_digest
        and evidence.rollback_readiness_digest == readiness.digest
    )
def rollback_is_ready(gate: ReleaseGate, evidence: EvidenceBundle, lease_not_after: datetime, now: datetime) -> bool:
    readiness = evidence.rollback_readiness
    return (
        readiness.status is RollbackStatus.READY
        and readiness.valid_until > now
        and lease_not_after <= readiness.valid_until
        and lease_not_after < readiness.retained_until
        and readiness.rollback_configuration_digest == gate.binding.rollback_configuration_digest
        and readiness.fallback_configuration_digest == gate.binding.fallback_configuration_digest
        and readiness.readiness_contract_digest == gate.binding.rollback_readiness_contract_digest
        and evidence.rollback_readiness_digest == readiness.digest
    )
def grant_within(gate: ReleaseGate, requested: AuthorityGrant) -> bool:
    radius = requested.limit.radius
    if radius.digest != gate.binding.evaluated_radius.digest: return False
    if gate.allowed_next.radius.digest != gate.binding.evaluated_radius.digest: return False
    requested_capabilities = set(requested.limit.envelope.capabilities)
    allowed_capabilities = set(gate.allowed_next.envelope.capabilities)
    return requested_capabilities.issubset(allowed_capabilities) if gate.allow_capability_narrowing else requested_capabilities == allowed_capabilities
def evidence_is_current(evidence: EvidenceBundle, now: datetime) -> bool:
    freshness = (evidence.monitorability, evidence.calibration, evidence.reviewer_capacity, evidence.budget, evidence.pairing)
    return evidence.valid_until > now and evidence.containment is ContainmentStatus.PASS and all(item is EvidenceFreshness.CURRENT for item in freshness) and not evidence.unresolved
def degraded_grant(gate: ReleaseGate, now: datetime) -> AuthorityGrant:
    return AuthorityGrant(limit=gate.degraded, lease_not_after=now)
def make_plan(
    *, disposition: DecisionState, grant: AuthorityGrant, reason: DecisionReason, page: bool = False, rollback: str | None = None, fallback: str | None = None,
) -> DecisionPlan:
    return DecisionPlan(disposition=disposition, resulting_grant=grant, operator_page_requested=page, reasons=(reason,), rollback_configuration_digest=rollback, fallback_configuration_digest=fallback)
def emergency_plan(
    gate: ReleaseGate, evidence: EvidenceBundle, *, response: DecisionState, reason: DecisionReason, page: bool, now: datetime,
) -> DecisionPlan:
    if response is DecisionState.ROLLBACK and rollback_is_ready(gate, evidence, now, now):
        return make_plan(disposition=DecisionState.ROLLBACK, grant=degraded_grant(gate, now), reason=reason, page=page, rollback=gate.binding.rollback_configuration_digest, fallback=gate.binding.fallback_configuration_digest)
    return make_plan(disposition=DecisionState.STOP, grant=degraded_grant(gate, now), reason=reason, page=page)
def classify(
    gate: ReleaseGate, evidence: EvidenceBundle, prior_state: AuthorityState,
    requested_grant: AuthorityGrant | None, *, now: datetime,
) -> DecisionPlan:
    if not evidence_matches_gate(gate, evidence):
        return make_plan(disposition=DecisionState.ABSTAIN_DEGRADE, grant=degraded_grant(gate, now), reason=DecisionReason.POLICY_MISMATCH)
    if isinstance(evidence.adverse_condition, VerifiedHarm):
        return emergency_plan(gate, evidence, response=gate.verified_harm_disposition, reason=DecisionReason.VERIFIED_HARM, page=True, now=now)
    if evidence.containment is ContainmentStatus.BREACH:
        return emergency_plan(gate, evidence, response=gate.containment_breach_disposition, reason=DecisionReason.CONTAINMENT_BREACH, page=False, now=now)
    if not evidence_is_current(evidence, now):
        reason = DecisionReason.EVIDENCE_EXPIRED if evidence.valid_until <= now else DecisionReason.EVIDENCE_UNAVAILABLE
        return make_plan(disposition=DecisionState.ABSTAIN_DEGRADE, grant=degraded_grant(gate, now), reason=reason)
    if evidence.quality is not PairedQualityStatus.CLEAR:
        if prior_state.active_grant.lease_not_after <= now:
            return make_plan(disposition=DecisionState.ABSTAIN_DEGRADE, grant=degraded_grant(gate, now), reason=DecisionReason.PRIOR_LEASE_EXPIRED)
        reason = DecisionReason.QUALITY_REGRESSION if evidence.quality is PairedQualityStatus.REGRESSION else DecisionReason.QUALITY_INCONCLUSIVE
        return make_plan(disposition=gate.uncertain_quality_disposition, grant=prior_state.active_grant, reason=reason)
    if requested_grant is None:
        raise AuthorizationRejected("clear evidence requires an exact release grant")
    if requested_grant.lease_not_after <= now:
        raise AuthorizationRejected("release lease must extend beyond decision time")
    if requested_grant.lease_not_after > evidence.valid_until:
        raise AuthorizationRejected("release lease exceeds evidence validity")
    if not rollback_is_ready(gate, evidence, requested_grant.lease_not_after, now):
        return make_plan(disposition=DecisionState.ABSTAIN_DEGRADE, grant=degraded_grant(gate, now), reason=DecisionReason.ROLLBACK_UNREADY)
    if not grant_within(gate, requested_grant):
        raise AuthorizationRejected("requested grant exceeds evaluated radius")
    return make_plan(disposition=DecisionState.RELEASE, grant=requested_grant, reason=DecisionReason.PAIRED_CLEAR)
def assert_human_authority(
    gate: ReleaseGate, principal: AuthenticatedPrincipal,
    disposition: DecisionState, now: datetime,
) -> None:
    if principal.principal_kind is not PrincipalKind.HUMAN:
        raise AuthorizationRejected(
            "instrument identities cannot issue decisions")
    if principal.valid_until <= now:
        raise AuthorizationRejected("authenticated principal has expired")
    authorization = next(
        (item for item in gate.authorizations if item.disposition is disposition),
        None,
    )
    if (
        authorization is None
        or principal.subject_id not in authorization.authorized_owner_ids
    ):
        raise AuthorizationRejected(
            "principal is not listed for this disposition")
    if not set(authorization.required_roles).issubset(set(principal.roles)):
        raise AuthorizationRejected("principal lacks the required role")
def assert_request_bound(
    request: SignedDecisionRequest, principal: AuthenticatedPrincipal,
    gate: ReleaseGate, evidence: EvidenceBundle, prior_state: AuthorityState,
    now: datetime,
) -> None:
    if request.issued_at > now or request.valid_until <= now:
        raise AuthorizationRejected("request is not currently valid")
    if (
        request.owner_subject_id != principal.subject_id
        or request.policy_digest != gate.digest
        or request.binding_digest != gate.binding.digest
        or request.evaluated_radius_digest != gate.binding.evaluated_radius.digest
        or request.evidence_bundle_digest != evidence.digest
        or request.prior_state_digest != prior_state.digest
    ):
        raise AuthorizationRejected("request does not bind this decision")
    if (
        request.requested_grant is not None
        and request.requested_grant.limit.radius.digest
        != request.evaluated_radius_digest
    ):
        raise AuthorizationRejected("release request exceeds evaluated radius")
def assert_effect_authorized(
    context: AuthenticatedEffectContext,
    active_state: AuthorityState,
    now: datetime,
) -> None:
    require_aware(now); grant = active_state.active_grant
    if (
        context.authority_receipt_digest != active_state.active_receipt_digest
        or context.revocation_epoch != active_state.revocation_epoch
    ):
        raise AuthorizationRejected("effect context is stale")
    if (
        grant.lease_not_after <= now
        or context.runtime_radius.digest != grant.limit.radius.digest
    ):
        raise AuthorizationRejected("effect exceeds active radius")
    if (
        context.capability not in grant.limit.envelope.capabilities
        or context.permission not in grant.limit.envelope.permissions
    ):
        raise AuthorizationRejected("effect exceeds active authority")
def contain_breach(
    *, gate: ReleaseGate, evidence: EvidenceBundle, prior_state: AuthorityState,
    verifier: TrustedAuthorityBoundary, outcome_correlation_id: NonEmptyText,
) -> ContainmentCommand:
    now = verifier.decision_time(); require_aware(now)
    verifier.verify_preregistered_policy(gate, now)
    verifier.verify_containment_breach(evidence, gate, now)
    if evidence.containment is not ContainmentStatus.BREACH:
        raise AuthorizationRejected(
            "denial transition requires an authenticated breach")
    return ContainmentCommand(
        expected_prior_state_digest=prior_state.digest,
        expected_sequence=prior_state.sequence,
        expected_revocation_epoch=prior_state.revocation_epoch,
        policy_digest=gate.digest,
        evidence_bundle_digest=evidence.digest,
        resulting_grant=degraded_grant(gate, now),
        resulting_revocation_epoch=prior_state.revocation_epoch + 1,
        reason=DecisionReason.CONTAINMENT_BREACH, observed_at=now,
        outcome_correlation_id=outcome_correlation_id,
    )
def decide_release(
    *, gate: ReleaseGate, evidence: EvidenceBundle, prior_state: AuthorityState,
    principal: AuthenticatedPrincipal, request: SignedDecisionRequest,
    verifier: TrustedAuthorityBoundary,
    outcome_correlation_id: NonEmptyText,
) -> DecisionCommand:
    """Return a command for durable compare-and-set; never mutate authority."""
    now = verifier.decision_time(); require_aware(now)
    verifier.verify_preregistered_policy(gate, now)
    verifier.verify_evidence_bundle(evidence, gate, now)
    verifier.verify_principal(principal, now)
    assert_request_bound(request, principal, gate, evidence, prior_state, now)
    verifier.verify_request(request, principal, gate, evidence, prior_state, now)
    plan = classify(gate, evidence, prior_state, request.requested_grant, now=now)
    assert_human_authority(gate, principal, plan.disposition, now)
    if request.requested_disposition != plan.disposition:
        raise AuthorizationRejected("signed disposition differs from policy")
    if plan.disposition is DecisionState.RELEASE:
        if (
            request.requested_grant is None
            or request.requested_grant.digest != plan.resulting_grant.digest
        ):
            raise AuthorizationRejected(
                "signed release grant differs from policy")
    elif request.expected_limit_digest != plan.resulting_grant.limit.digest:
        raise AuthorizationRejected(
            "signed non-release limit differs from policy")
    revoking = plan.disposition in {
        DecisionState.ROLLBACK, DecisionState.ABSTAIN_DEGRADE, DecisionState.STOP}
    next_epoch = prior_state.revocation_epoch + int(revoking)
    return DecisionCommand(
        expected_prior_state_digest=prior_state.digest,
        expected_sequence=prior_state.sequence,
        expected_revocation_epoch=prior_state.revocation_epoch,
        policy_digest=gate.digest, binding_digest=gate.binding.digest,
        evaluated_radius_digest=gate.binding.evaluated_radius.digest,
        evidence_bundle_digest=evidence.digest,
        owner_subject_id=principal.subject_id, request_digest=request.digest,
        disposition=plan.disposition, resulting_grant=plan.resulting_grant,
        resulting_revocation_epoch=next_epoch,
        operator_page_requested=plan.operator_page_requested,
        reasons=plan.reasons,
        outcome_correlation_id=outcome_correlation_id,
        idempotency_key=request.idempotency_key,
        rollback_configuration_digest=plan.rollback_configuration_digest,
        fallback_configuration_digest=plan.fallback_configuration_digest,
    )
