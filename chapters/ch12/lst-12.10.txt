def digest(seed: str = "a") -> str:
    if len(seed) == 1 and seed in "0123456789abcdef":
        value = seed * 64
    else:
        value = hashlib.sha256(seed.encode("utf-8")).hexdigest()
    return f"sha256:{value}"


def known(identifier: str, revision: str) -> KnownIdentity:
    return KnownIdentity(
        identifier=identifier,
        revision=revision,
        digest=digest(),
    )


def requested_route(revision: str) -> KnownIdentity:
    return KnownIdentity(
        identifier="requested-route",
        revision=revision,
        identity_basis="operator_configured_exact_value",
    )


def provider_checkpoint(
    revision: str,
    *,
    verified_on: date = date(2026, 7, 1),
    expires_on: date | None = date(2027, 7, 1),
) -> KnownIdentity:
    return KnownIdentity(
        identifier="hosted-backend",
        revision=revision,
        identity_basis="provider_observed_revision",
        identity_evidence=IdentityEvidenceReceipt(
            claimed_identifier="hosted-backend",
            claimed_revision=revision,
            verifier_boundary="identity-registry",
            method="registered_immutable_release",
            evidence_locator="registry://hosted-backend",
            evidence_digest=digest("b"),
            verified_on=verified_on,
            expires_on=expires_on,
        ),
    )


def run_bound_evidence(
    suffix: str,
    backend: KnownIdentity,
    *,
    observed_on: date = date(2026, 7, 20),
) -> RunBoundBackendEvidence:
    return RunBoundBackendEvidence(
        run_id=f"run-{suffix}",
        request_id=f"request-{suffix}",
        claimed_backend_identifier=backend.identifier,
        claimed_backend_revision=backend.revision,
        attestation_method="gateway_attested",
        evidence_locator=f"gateway://receipts/run-{suffix}",
        evidence_digest=digest(f"run-bound-{suffix}"),
        observed_on=observed_on,
    )

        configured_backend_identity=backend_identity,
        reasoning_effort=known("effort-policy", "v2"),
        prompt_policy=known("prompt-policy", "v9"),
        retrieval=known("retrieval-policy", "disabled-v1"),
        tools=known("tool-policy", "disabled-v1"),
        memory=known("memory-policy", "disabled-v1"),
        topology=known("configured-topology", "single-route-v3"),
        routing_policy=known("routing-policy", "primary-only-v2"),
        in_loop_evaluators=known(
            "in-loop-evaluator-bundle",
            "disabled-v1",
        ),
        cache=known("cache-policy", "scope-v1"),
        budgets=known("budget-policy", "v4"),
        fallback=known("fallback-policy", "disabled-v1"),
        tokenizer=known("tokenizer", "revision-8"),
        sampling=known("sampling-policy", "greedy-v2"),
        adapter=NotApplicable(reason="no adapter is loaded"),
        locale=NotApplicable(reason="no locale surface exists"),
        runtime_sandbox=known("sandbox", "revision-3"),
        hardware_kernel=UnknownIdentity(
            provider_or_boundary="hosted-runtime",
            observed_on=date(2026, 7, 20),
            public_identity="provider-owned-numerical-kernel",
            reason="the present numerical kernel is not disclosed",
        ),
        external_services=NotApplicable(reason="no external service exists"),
        concurrency_scheduler=NotApplicable(
            reason="no concurrent scheduler exists",
        ),
        conditional_presence={
            "tokenizer": "present",
            "sampling": "present",
            "adapter": "absent",
            "locale": "absent",
            "runtime_sandbox": "present",
            "hardware_kernel": "present",
            "external_services": "absent",

def fully_known_runtime(
    requested_route_value: KnownIdentity,
    backend_identity: KnownIdentity | UnknownIdentity | None = None,
) -> RuntimeConfiguration:
    configured = runtime(requested_route_value, backend_identity)
    payload = configured.model_dump(
        mode="python",
        exclude={
            "configuration_digest",
            "runtime_verification_digest",
            "all_behavior_known",
        },
    )
    payload["hardware_kernel"] = known("numerical-kernel", "v3")
    return RuntimeConfiguration.model_validate(payload)


def population() -> PopulationIdentity:
    return PopulationIdentity(
        dataset_population=known("case-population", "slice-2026-07"),
    )


def evaluators() -> EvaluatorContract:
    return EvaluatorContract(
        evaluators=known("evaluator-contract", "rubric-v5"),
    )


def controlled_plan(
    pop: PopulationIdentity,
    evaluator_contract: EvaluatorContract,
) -> ComparisonPlan:
    baseline_route = requested_route("route-v1")
    candidate_route = requested_route("route-v2")
    baseline_runtime = fully_known_runtime(baseline_route)
    candidate_runtime = fully_known_runtime(candidate_route)
    held = tuple(
        field for field in ALL_RUNTIME_SURFACES if field != "requested_route"
    ) + ("dataset_population", "evaluators")
    return ComparisonPlan(
        mode="controlled",
        configuration_digests={
            "baseline": baseline_runtime.configuration_digest,
            "candidate": candidate_runtime.configuration_digest,
        },
        population=pop,
        evaluators=evaluator_contract,
        varied_fields=("requested_route",),
        held_constant_fields=held,
        declared_arm_values={
            "baseline": {"requested_route": baseline_route},
            "candidate": {"requested_route": candidate_route},
        },
    )


def observation(
    suffix: str = "a",
    observed_on: date = date(2026, 7, 20),
    *,
    realized_route: str = "primary",
    realized_backend_identity: KnownIdentity | UnknownIdentity | None = None,
    execution_path_digest: str | None = None,
    fallback_decisions: tuple[str, ...] = ("primary-selected",),
    service_invocations: tuple[str, ...] = (),
) -> RunObservationReceipt:
    if realized_backend_identity is None:
        realized_backend_identity = provider_checkpoint("backend-v1")
    backend_evidence = (
        run_bound_evidence(
            suffix,
            realized_backend_identity,
            observed_on=observed_on,
        )
        if isinstance(realized_backend_identity, KnownIdentity)
        else None
    )
    return RunObservationReceipt(
        run_id=f"run-{suffix}",
        request_id=f"request-{suffix}",
        randomness_provenance=f"provider-request-{suffix}",
        realized_route=realized_route,
        realized_backend_identity=realized_backend_identity,
        run_bound_backend_evidence=backend_evidence,
        execution_path_digest=(
            execution_path_digest or digest(f"execution-{suffix}")
        ),
        fallback_decisions=fallback_decisions,
        service_invocations=service_invocations,
        realized_output_digest=digest(suffix),
        observed_effects_digest=digest("e"),
        observed_on=observed_on,
    )


def controlled_subject(
    arm: Literal["baseline", "candidate"] = "baseline",
    *,
    verified_on: date = date(2026, 7, 1),
) -> EvaluationSubject:
    pop = population()
    evaluator_contract = evaluators()
    plan = controlled_plan(pop, evaluator_contract)
    route = plan.declared_arm_values[arm]["requested_route"]
    assert isinstance(route, KnownIdentity)
    backend = provider_checkpoint(
        "backend-v1",
        verified_on=verified_on,
    )
    configured = fully_known_runtime(route, backend)
    return EvaluationSubject(
        runtime=configured,
        population=pop,
        evaluators=evaluator_contract,
        comparison_plan=plan,
        arm=arm,
        evaluated_on=date(2026, 7, 20),
        run_observation=observation(arm),
    )

def runtime_input(configured: RuntimeConfiguration) -> dict[str, object]:
    return configured.model_dump(
        mode="python",
        exclude={
            "configuration_digest",
            "runtime_verification_digest",
            "all_behavior_known",
        },
    )


def subject_input(subject: EvaluationSubject) -> dict[str, object]:
    plan = subject.comparison_plan
    return {
        "runtime": runtime_input(subject.runtime),
        "population": subject.population.model_dump(
            mode="python",
            exclude={"population_digest"},
        ),
        "evaluators": subject.evaluators.model_dump(
            mode="python",
            exclude={"evaluator_digest"},
        ),
        "comparison_plan": {
            "mode": plan.mode,
            "configuration_digests": plan.configuration_digests,
            "population": plan.population.model_dump(
                mode="python",
                exclude={"population_digest"},
            ),
            "evaluators": plan.evaluators.model_dump(
                mode="python",
                exclude={"evaluator_digest"},
            ),
            "varied_fields": plan.varied_fields,
            "held_constant_fields": plan.held_constant_fields,
            "declared_arm_values": plan.declared_arm_values,
        },
        "arm": subject.arm,
        "evaluated_on": subject.evaluated_on,
        "run_observation": subject.run_observation.model_dump(
            mode="python",
            exclude={"observation_digest"},
        ),
    }


def with_cache(
    subject: EvaluationSubject,
    cache: KnownIdentity | UnknownIdentity,
) -> EvaluationSubject:
    payload = subject_input(subject)
    configured = dict(cast(dict[str, object], payload["runtime"]))
    configured["cache"] = cache
    payload["runtime"] = configured
    return EvaluationSubject.model_validate(payload)


def forged_configured_backend(
    backend: KnownIdentity,
    forgery_path: Literal["model_copy", "model_construct"],
) -> KnownIdentity:
    update = {
        "identity_basis": "operator_configured_exact_value",
        "identity_evidence": None,
    }
    if forgery_path == "model_copy":
        return backend.model_copy(update=update)
    payload = backend.model_dump(
        mode="python",
        exclude_computed_fields=True,
    )
    payload.update(update)
    return KnownIdentity.model_construct(**payload)


def subject_with_forged_configured_backend(
    subject: EvaluationSubject,
    forgery_path: Literal["model_copy", "model_construct"],
) -> EvaluationSubject:
    backend = subject.runtime.configured_backend_identity
    assert isinstance(backend, KnownIdentity)
    forged_backend = forged_configured_backend(backend, forgery_path)
    forged_runtime = subject.runtime.model_copy(
        update={"configured_backend_identity": forged_backend},
    )
    return subject.model_copy(update={"runtime": forged_runtime})


def assign_item(mapping: object, key: object, value: object) -> None:
    cast(dict[object, object], mapping)[key] = value


def delete_item(mapping: object, key: object) -> None:
    del cast(dict[object, object], mapping)[key]

def test_nested_mappings_are_deeply_immutable_and_order_stable() -> None:
    subject = controlled_subject()
    plan = subject.comparison_plan
    configuration_digest = subject.configuration_digest
    comparison_plan_digest = plan.comparison_plan_digest
    receipt_digest = subject.receipt_digest
    baseline_values = plan.declared_arm_values["baseline"]

    attempts = (
        (
            subject.runtime.conditional_presence,
            "adapter",
            "present",
        ),
        (
            plan.configuration_digests,
            "baseline",
            digest("mutated-configuration"),
        ),
        (
            plan.declared_arm_values,
            "baseline",
            {},
        ),
        (
            baseline_values,
            "requested_route",
            known("mutated-route", "v1"),
        ),
    )
    for mapping, key, value in attempts:
        with pytest.raises(TypeError):
            assign_item(mapping, key, value)
        with pytest.raises(TypeError):
            delete_item(mapping, key)

    assert subject.configuration_digest == configuration_digest
    assert plan.comparison_plan_digest == comparison_plan_digest
    assert subject.receipt_digest == receipt_digest

    runtime_payload = runtime_input(subject.runtime)
    runtime_payload["conditional_presence"] = dict(
        reversed(tuple(subject.runtime.conditional_presence.items())),
    )
    reconstructed_runtime = RuntimeConfiguration.model_validate(
        runtime_payload,
    )
    assert reconstructed_runtime.configuration_digest == configuration_digest

    plan_payload = dict(
        cast(
            dict[str, object],
            subject_input(subject)["comparison_plan"],
        ),
    )
    plan_payload["configuration_digests"] = dict(
        reversed(tuple(plan.configuration_digests.items())),
    )
    plan_payload["declared_arm_values"] = {
        arm: dict(reversed(tuple(values.items())))
        for arm, values in reversed(tuple(plan.declared_arm_values.items()))
    }
    reconstructed_plan = ComparisonPlan.model_validate(plan_payload)
    assert reconstructed_plan.comparison_plan_digest == comparison_plan_digest

def adapter_treatment_subject(
    arm: Literal["baseline", "candidate"],
) -> EvaluationSubject:
    pop = population()
    evaluator_contract = evaluators()
    baseline_value = NotApplicable(reason="no adapter is loaded")
    candidate_value = known("adapter", "adapter-v2")
    held = tuple(
        field for field in ALL_RUNTIME_SURFACES if field != "adapter"
    ) + ("dataset_population", "evaluators")
    runtimes: dict[str, RuntimeConfiguration] = {}
    for current_arm, value in (
        ("baseline", baseline_value),
        ("candidate", candidate_value),
    ):
        payload = runtime_input(
            runtime(requested_route("route-v1")),
        )
        payload["hardware_kernel"] = known("numerical-kernel", "v3")
        payload["adapter"] = value
        presence = dict(payload["conditional_presence"])
        presence["adapter"] = (
            "absent" if current_arm == "baseline" else "present"
        )
        payload["conditional_presence"] = presence
        runtimes[current_arm] = RuntimeConfiguration.model_validate(
            payload,
        )
    plan = ComparisonPlan(
        mode="controlled",
        configuration_digests={
            key: value.configuration_digest
            for key, value in runtimes.items()
        },
        population=pop,
        evaluators=evaluator_contract,
        varied_fields=("adapter",),
        held_constant_fields=held,
        declared_arm_values={
            "baseline": {"adapter": baseline_value},
            "candidate": {"adapter": candidate_value},
        },
    )
    return EvaluationSubject(
        runtime=runtimes[arm],
        population=pop,
        evaluators=evaluator_contract,
        comparison_plan=plan,
        arm=arm,
        evaluated_on=date(2026, 7, 20),
        run_observation=observation(f"adapter-{arm}"),
    )

def controlled_pair_with_candidate_cache(
    cache: KnownIdentity | UnknownIdentity,
) -> tuple[EvaluationSubject, EvaluationSubject]:
    pop = population()
    evaluator_contract = evaluators()
    baseline_route = requested_route("route-v1")
    candidate_route = requested_route("route-v2")
    baseline_runtime = fully_known_runtime(baseline_route)
    payload = runtime_input(fully_known_runtime(candidate_route))
    payload["cache"] = cache
    candidate_runtime = RuntimeConfiguration.model_validate(payload)
    held = tuple(
        field for field in ALL_RUNTIME_SURFACES if field != "requested_route"
    ) + ("dataset_population", "evaluators")
    plan = ComparisonPlan(
        mode="controlled",
        configuration_digests={
            "baseline": baseline_runtime.configuration_digest,
            "candidate": candidate_runtime.configuration_digest,
        },
        population=pop,
        evaluators=evaluator_contract,
        varied_fields=("requested_route",),
        held_constant_fields=held,
        declared_arm_values={
            "baseline": {"requested_route": baseline_route},
            "candidate": {"requested_route": candidate_route},
        },
    )
    subjects = tuple(
        EvaluationSubject(
            runtime=(
                baseline_runtime if arm == "baseline" else candidate_runtime
            ),
            population=pop,
            evaluators=evaluator_contract,
            comparison_plan=plan,
            arm=arm,
            evaluated_on=date(2026, 7, 20),
            run_observation=observation(arm),
        )
        for arm in ("baseline", "candidate")
    )
    return cast(tuple[EvaluationSubject, EvaluationSubject], subjects)
