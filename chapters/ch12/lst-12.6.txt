def non_blank(value: str) -> str:
    normalized = value.strip()
    if not normalized:
        raise ValueError("value must be non-blank")
    return normalized


def immutable_revision(value: str) -> str:
    normalized = non_blank(value)
    if normalized.lower() in {"latest", "current", "production", "default"}:
        raise ValueError("revision must be immutable")
    return normalized


def finite_decimal(value: Decimal) -> Decimal:
    if not value.is_finite():
        raise ValueError("decimal must be finite")
    return value


def utc_instant(value: datetime) -> datetime:
    if value.utcoffset() != timedelta(0):
        raise ValueError("timestamps must be timezone-aware UTC")
    return value

class FrozenModel(BaseModel):
    model_config = ConfigDict(
        extra="forbid",
        frozen=True,
        strict=True,
        revalidate_instances="always",
    )

    def validated_copy(self, *, update: Mapping[str, object]) -> Self:
        payload = self.model_dump(mode="python", exclude_computed_fields=True)
        payload.update(update)
        return type(self).model_validate(payload)


class ReceiptBound(Protocol):
    receipt_digest: str
    evaluator_digest: str


def receipt_binding(value: ReceiptBound) -> tuple[str, str]:
    return (value.receipt_digest, value.evaluator_digest)


def canonical_sha256(value: object) -> str:
    payload = json.dumps(
        value,
        ensure_ascii=True,
        separators=(",", ":"),
        sort_keys=True,
    ).encode("utf-8")
    return f"sha256:{hashlib.sha256(payload).hexdigest()}"


class Interval(FrozenModel):
    lower: FiniteDecimal
    upper: FiniteDecimal

    @model_validator(mode="after")
    def ordered(self) -> Self:
        if self.lower > self.upper:
            raise ValueError("interval lower bound exceeds upper bound")
        return self

    def contains(self, value: Decimal) -> bool:
        return self.lower <= value <= self.upper

class RequirementSpec(FrozenModel):
    requirement_id: NonBlank
    predicate: NonBlank
    predicate_or_oracle_revision: ImmutableRevision
    admitted_evidence_kinds: tuple[EvidenceKind, ...]
    violation_policy: ViolationPolicy
    robustness_component: RobustnessComponentName | None = None

    @model_validator(mode="after")
    def require_admitted_evidence_kinds(self) -> Self:
        if not self.admitted_evidence_kinds:
            raise ValueError("requirement must admit at least one evidence kind")
        if len(set(self.admitted_evidence_kinds)) != len(
            self.admitted_evidence_kinds,
        ):
            raise ValueError("admitted evidence kinds must be unique")
        return self


class EvidenceReference(FrozenModel):
    receipt_digest: Sha256
    planned_case_manifest_digest: Sha256
    case_digest: Sha256
    requirement_id: NonBlank
    robustness_component: RobustnessComponentName | None = None
    predicate_or_oracle_revision: ImmutableRevision
    locator: NonBlank
    revision: ImmutableRevision
    evidence_digest: Sha256
    evidence_kind: EvidenceKind
    observed_on: UtcInstant
    availability: Literal["OBSERVED", "UNAVAILABLE"]
    assessment: Literal["SATISFIED", "VIOLATED", "UNAVAILABLE"]
    effect_receipt_digest: Sha256 | None = None

    @model_validator(mode="after")
    def bind_availability_to_receipt(self) -> Self:
        if self.availability == "UNAVAILABLE":
            if (
                self.assessment != "UNAVAILABLE"
                or self.evidence_kind != "absence_receipt"
                or self.effect_receipt_digest is not None
            ):
                raise ValueError("unavailable evidence requires an absence receipt")
            return self
        if self.evidence_kind == "absence_receipt":
            raise ValueError("an absence receipt requires unavailable evidence")
        if self.assessment == "UNAVAILABLE":
            raise ValueError("observed evidence needs an assessment")
        if self.assessment == "VIOLATED":
            if self.evidence_kind not in {"predicate_receipt", "effect_receipt"}:
                raise ValueError(
                    "a violation requires a structurally admitted predicate/effect receipt",
                )
            if (
                self.evidence_kind == "effect_receipt"
                and self.effect_receipt_digest is not None
            ):
                raise ValueError("an effect receipt cannot link to another effect receipt")
        elif self.effect_receipt_digest is not None:
            raise ValueError("only violated evidence carries an effect receipt")
        return self

class CostAccounting(FrozenModel):
    basis: AccountingBasis
    generation_included: bool
    retries_included: bool
    evaluator_calls_included: bool
    infrastructure_included: bool
    human_adjudication_included: bool
    storage_included: bool
    queueing_included: bool
    fixed_cost_allocation_revision: ImmutableRevision | None = None
    excluded_fixed_costs: tuple[NonBlank, ...] = ()

    @model_validator(mode="after")
    def require_declared_accounting_basis(self) -> Self:
        if not (self.retries_included and self.evaluator_calls_included):
            raise ValueError("retry and evaluator-call costs must be included")
        included = (
            self.generation_included,
            self.retries_included,
            self.evaluator_calls_included,
            self.infrastructure_included,
            self.human_adjudication_included,
            self.storage_included,
            self.queueing_included,
        )
        if self.basis == "FULLY_LOADED":
            if not all(included) or self.fixed_cost_allocation_revision is None:
                raise ValueError(
                    "fully loaded accounting requires all cost classes and allocation revision",
                )
            if self.excluded_fixed_costs:
                raise ValueError("fully loaded accounting cannot exclude fixed costs")
        elif (
            self.fixed_cost_allocation_revision is not None
            or not self.excluded_fixed_costs
        ):
            raise ValueError(
                "marginal accounting declares no allocation and names excluded fixed costs",
            )
        return self

class EvaluatorContractSpec(FrozenModel):
    contract_id: NonBlank
    contract_revision: ImmutableRevision
    family: MetricFamily
    claimed_property: NonBlank
    scope: NonBlank
    population: NonBlank
    observation_boundary: NonBlank
    requirements: tuple[RequirementSpec, ...]
    exclusions: tuple[NonBlank, ...]
    metric_revision: ImmutableRevision
    evaluator_revision: ImmutableRevision
    evaluator_digest: Sha256
    rubric_or_oracle_revision: ImmutableRevision
    unit: NonBlank
    direction: MetricDirection
    estimator_revision: ImmutableRevision
    missing_data_rule: MissingDataRule
    uncertainty_method: NonBlank
    blind_spots: tuple[NonBlank, ...]
    common_mode_dependencies: tuple[NonBlank, ...]
    decision_rule: NonBlank
    adjudication_path: tuple[NonBlank, ...]
    decision_semantics: DecisionSemantics
    cost_accounting: CostAccounting | None = None
    currency: CurrencyCode | None = None

    @model_validator(mode="after")
    def validate_contract(self) -> Self:
        declarations = (
            self.requirements,
            self.exclusions,
            self.blind_spots,
            self.common_mode_dependencies,
            self.adjudication_path,
        )
        if any(not declaration for declaration in declarations):
            raise ValueError("contract declarations must be explicit")
        identifiers = tuple(item.requirement_id for item in self.requirements)
        if len(set(identifiers)) != len(identifiers):
            raise ValueError("requirement identifiers must be unique")
        expected_semantics = DECISION_SEMANTICS_BY_FAMILY[self.family]
        if self.decision_semantics != expected_semantics:
            if self.family == "CONTAINMENT":
                raise ValueError("containment must be non-compensatory")
            raise ValueError(
                f"{self.family} decision semantics must be {expected_semantics}",
            )
        components = tuple(
            item.robustness_component
            for item in self.requirements
            if item.robustness_component is not None
        )
        if self.family == "ROBUSTNESS":
            if set(components) != {"h", "b", "f", "ell", "c", "o"}:
                raise ValueError(
                    "robustness requires one requirement for each typed component",
                )
            if len(components) != len(set(components)):
                raise ValueError("robustness components must be unique")
        elif components:
            raise ValueError("robustness components belong only to robustness")
        if self.family in {"OPERATIONAL_COST", "ROBUSTNESS"}:
            if self.cost_accounting is None or self.currency is None:
                raise ValueError(
                    "cost-bearing contracts require accounting and currency",
                )
        elif self.cost_accounting is not None or self.currency is not None:
            raise ValueError(
                "accounting and currency belong only to cost-bearing contracts",
            )
        return self

class StochasticObservation(FrozenModel):
    kind: Literal["stochastic"] = "stochastic"
    state: Literal["ESTIMATED", "INCONCLUSIVE", "NOT_ESTIMABLE", "ABSTAIN"]
    estimate: FiniteDecimal | None = None
    interval: Interval | None = None
    reason: NonBlank | None = None

    @model_validator(mode="after")
    def enforce_state_semantics(self) -> Self:
        paired = (self.estimate is None) == (self.interval is None)
        if self.state == "ESTIMATED":
            if self.estimate is None or self.interval is None:
                raise ValueError("estimated observations require estimate and interval")
        elif self.state == "INCONCLUSIVE":
            if self.reason is None:
                raise ValueError("inconclusive observations require a reason")
            if not paired:
                raise ValueError("diagnostic estimate and interval are paired")
        elif (
            self.reason is None
            or self.estimate is not None
            or self.interval is not None
        ):
            raise ValueError("not-estimable and abstained observations carry reason only")
        if self.estimate is not None:
            if self.interval is None or not self.interval.contains(self.estimate):
                raise ValueError("estimate must lie inside its interval")
        return self

class RobustnessComponent(FrozenModel):
    state: Literal["ESTIMATED", "INCONCLUSIVE", "NOT_ESTIMABLE", "ABSTAIN"]
    value: FiniteDecimal | None = None
    interval: Interval | None = None
    uncertainty_method: NonBlank
    missingness: Missingness
    reason: NonBlank | None = None

    @model_validator(mode="after")
    def enforce_component_state(self) -> Self:
        paired = (self.value is None) == (self.interval is None)
        if self.state == "ESTIMATED":
            if self.value is None or self.interval is None:
                raise ValueError("estimated component requires value and interval")
            if self.missingness.state != "COMPLETE":
                raise ValueError(
                    "estimated component requires COMPLETE missingness",
                )
            if self.reason is not None:
                raise ValueError("estimated component cannot carry a failure reason")
        elif self.state == "INCONCLUSIVE":
            if self.reason is None:
                raise ValueError("inconclusive component requires a reason")
            if not paired:
                raise ValueError("diagnostic component value and interval are paired")
            if self.value is not None and self.missingness.state != "PARTIAL":
                raise ValueError(
                    "inconclusive component missingness contradicts its numeric payload",
                )
            if self.value is None and self.missingness.state not in {
                "COMPLETE",
                "PARTIAL",
            }:
                raise ValueError(
                    "inconclusive component requires COMPLETE or PARTIAL missingness",
                )
        elif self.state == "NOT_ESTIMABLE":
            if (
                self.reason is None
                or self.value is not None
                or self.interval is not None
                or self.missingness.state != "UNAVAILABLE"
            ):
                raise ValueError(
                    "not-estimable component requires no value or interval "
                    "and UNAVAILABLE missingness",
                )
        elif (
            self.reason is None
            or self.value is not None
            or self.interval is not None
            or self.missingness.state != "ABSTAINED"
        ):
            raise ValueError(
                "abstained component requires no value or interval "
                "and ABSTAINED missingness",
            )
        if self.value is not None:
            if self.interval is None or not self.interval.contains(self.value):
                raise ValueError("component value must lie inside its interval")
        return self

class ExecutionRecord(FrozenModel):
    state: Literal["COMPLETED", "EVALUATOR_ERROR"]
    failure_class: FailureClass | None = None
    failure_detail: NonBlank | None = None
    attempts: PositiveInt
    elapsed_ms: NonNegativeInt
    incurred_evaluator_cost: CostCharge

    @model_validator(mode="after")
    def retain_error_provenance(self) -> Self:
        failed = self.state == "EVALUATOR_ERROR"
        has_failure_class = self.failure_class is not None
        has_failure_detail = self.failure_detail is not None
        if has_failure_class != has_failure_detail:
            raise ValueError("failure class and detail must be jointly present")
        if failed != has_failure_class:
            raise ValueError("execution state and failure provenance disagree")
        return self


class EvaluatorResult(FrozenModel):
    receipt_digest: Sha256
    evaluator_digest: Sha256
    contract: EvaluatorContractSpec
    planned_case_manifest_digest: Sha256
    case_id: NonBlank
    case_digest: Sha256
    case_slices: tuple[NonBlank, ...]
    evidence: tuple[EvidenceReference, ...]
    execution: ExecutionRecord
    observation: Observation | None = None
    missing_data_disposition: MissingDataRule | None = None
    supersedes_result_digest: Sha256 | None = None
    correction_reason: NonBlank | None = None

    @model_validator(mode="after")
    def enforce_evidence_and_result_semantics(self) -> Self:
        if self.evaluator_digest != self.contract.evaluator_digest:
            raise ValueError("evaluator digest does not match contract")
        required = tuple(item.requirement_id for item in self.contract.requirements)
        assessed = tuple(item.requirement_id for item in self.evidence)
        if (
            not self.case_slices
            or set(assessed) != set(required)
        ):
            raise ValueError("every required identifier needs evidence")
        evidence_digests = tuple(item.evidence_digest for item in self.evidence)
        if len(evidence_digests) != len(set(evidence_digests)):
            raise ValueError("evidence digests must be unique")
        requirements = {
            item.requirement_id: item for item in self.contract.requirements
        }
        for item in self.evidence:
            requirement = requirements[item.requirement_id]
            if item.receipt_digest != self.receipt_digest:
                raise ValueError("evidence receipt digest does not match result")
            if (
                item.planned_case_manifest_digest
                != self.planned_case_manifest_digest
            ):
                raise ValueError("evidence manifest digest does not match result")
            if item.case_digest != self.case_digest:
                raise ValueError("evidence case digest does not match result")
            if (
                item.predicate_or_oracle_revision
                != requirement.predicate_or_oracle_revision
            ):
                raise ValueError(
                    "evidence predicate/oracle revision is not admitted",
                )
            if item.evidence_kind not in requirement.admitted_evidence_kinds:
                raise ValueError("evidence kind is not admitted by requirement")
            if item.robustness_component != requirement.robustness_component:
                raise ValueError("evidence robustness component does not match requirement")
        if (self.supersedes_result_digest is None) != (self.correction_reason is None):
            raise ValueError("correction provenance requires both fields")

        if self.observation is not None:
            compatible_observation_types: dict[
                MetricFamily,
                type[FrozenModel] | tuple[type[FrozenModel], ...],
            ] = {
                "CONTAINMENT": (
                    DeterministicContainmentObservation,
                    ContainmentUnavailableObservation,
                ),
                "TASK_QUALITY": StochasticObservation,
                "CALIBRATION": StochasticObservation,
                "ROBUSTNESS": RobustnessObservation,
                "OPERATIONAL_COST": OperationalCostObservation,
                "OUTCOME": StochasticObservation,
            }
            if not isinstance(
                self.observation,
                compatible_observation_types[self.contract.family],
            ):
                raise ValueError(
                    "observation kind is incompatible with metric family",
                )

        if self.contract.family == "CONTAINMENT":
            claimed_violations = tuple(
                item
                for item in self.evidence
                if item.availability == "OBSERVED"
                and item.assessment == "VIOLATED"
            )
            violations: list[EvidenceReference] = []
            evidence_by_digest = {
                item.evidence_digest: item for item in self.evidence
            }
            for item in claimed_violations:
                requirement = requirements[item.requirement_id]
                if requirement.violation_policy == "PREDICATE_SUFFICIENT":
                    violations.append(item)
                    continue
                if item.evidence_kind == "effect_receipt":
                    violations.append(item)
                    continue
                if item.effect_receipt_digest is None:
                    raise ValueError(
                        "EFFECT_RECEIPT_REQUIRED violation is missing an effect-receipt link",
                    )
                if item.effect_receipt_digest == item.evidence_digest:
                    raise ValueError("effect-receipt linkage cannot be self-referential")
                effect = evidence_by_digest.get(item.effect_receipt_digest)
                if effect is None:
                    raise ValueError("effect-receipt linkage is dangling")
                if (
                    effect.evidence_kind != "effect_receipt"
                    or effect.availability != "OBSERVED"
                    or effect.assessment != "VIOLATED"
                    or effect.receipt_digest != item.receipt_digest
                    or effect.planned_case_manifest_digest
                    != item.planned_case_manifest_digest
                    or effect.case_digest != item.case_digest
                    or effect.requirement_id != item.requirement_id
                    or effect.predicate_or_oracle_revision
                    != item.predicate_or_oracle_revision
                    or effect.robustness_component != item.robustness_component
                ):
                    raise ValueError(
                        "effect-receipt linkage does not resolve to matching effect evidence",
                    )
                violations.append(item)
            if claimed_violations and not violations:
                raise ValueError(
                    "EFFECT_RECEIPT_REQUIRED violation lacks a resolvable effect receipt",
                )
            unavailable = any(
                item.availability == "UNAVAILABLE" for item in self.evidence
            )
            if violations:
                if not isinstance(
                    self.observation,
                    DeterministicContainmentObservation,
                ) or self.observation.status != "FAIL":
                    raise ValueError(
                        "structurally admitted, observed containment violation "
                        "requires FAIL",
                    )
                if unavailable:
                    if (
                        self.missing_data_disposition
                        != self.contract.missing_data_rule
                    ):
                        raise ValueError(
                            "unavailable evidence requires exact contract "
                            "disposition",
                        )
                elif self.missing_data_disposition is not None:
                    raise ValueError(
                        "complete evidence prohibits missing-data disposition",
                    )
                return self

            if self.execution.state == "EVALUATOR_ERROR":
                if self.observation is not None:
                    raise ValueError(
                        "evaluator errors without a violation cannot carry observations",
                    )
                if unavailable:
                    if (
                        self.missing_data_disposition
                        != self.contract.missing_data_rule
                    ):
                        raise ValueError(
                            "unavailable evidence requires exact contract "
                            "disposition",
                        )
                elif self.missing_data_disposition is not None:
                    raise ValueError(
                        "complete evidence prohibits missing-data disposition",
                    )
                return self

            if self.observation is None:
                raise ValueError("completed evaluation requires an observation")
            if isinstance(self.observation, ContainmentUnavailableObservation):
                if (
                    self.missing_data_disposition != self.contract.missing_data_rule
                    or not unavailable
                ):
                    raise ValueError("unavailable containment needs declared disposition")
                return self
            if not isinstance(self.observation, DeterministicContainmentObservation):
                raise ValueError("containment requires a deterministic observation")
            if unavailable:
                raise ValueError("containment pass or fail requires complete evidence")
            if self.observation.status == "PASS":
                if any(item.assessment != "SATISFIED" for item in self.evidence):
                    raise ValueError("containment pass requires all requirements satisfied")
                if any(
                    item.evidence_kind not in {"predicate_receipt", "effect_receipt"}
                    for item in self.evidence
                ):
                    raise ValueError(
                        "containment pass requires admitted predicate/effect receipts",
                    )
            else:
                raise ValueError(
                    "containment fail requires structurally admitted, observed violation evidence",
                )
            if self.missing_data_disposition is not None:
                raise ValueError("complete evidence prohibits missing-data disposition")
            return self

        if self.execution.state == "EVALUATOR_ERROR":
            if self.observation is not None:
                raise ValueError("evaluator errors cannot carry result observations")
            unavailable = any(
                item.availability == "UNAVAILABLE" for item in self.evidence
            )
            if unavailable:
                if (
                    self.missing_data_disposition
                    != self.contract.missing_data_rule
                ):
                    raise ValueError(
                        "unavailable evidence requires exact contract disposition",
                    )
            elif self.missing_data_disposition is not None:
                raise ValueError(
                    "complete evidence prohibits missing-data disposition",
                )
            return self

        if self.observation is None:
            raise ValueError("completed evaluation requires an observation")

        unavailable_evidence = tuple(
            item for item in self.evidence if item.availability == "UNAVAILABLE"
        )
        unavailable = bool(unavailable_evidence)
        if unavailable:
            if self.missing_data_disposition != self.contract.missing_data_rule:
                raise ValueError(
                    "unavailable evidence requires exact contract disposition",
                )
            if isinstance(self.observation, StochasticObservation):
                states = (self.observation.state,)
            else:
                states = ()
            if any(state == "ESTIMATED" for state in states):
                raise ValueError("unavailable evidence prohibits ESTIMATED")
        elif self.missing_data_disposition is not None:
            raise ValueError("complete evidence prohibits missing-data disposition")
        if self.contract.family == "ROBUSTNESS":
            if not isinstance(self.observation, RobustnessObservation):
                raise ValueError("robustness requires all six typed components")
            components = {
                component.component: component
                for component in (
                    self.observation.h,
                    self.observation.b,
                    self.observation.f,
                    self.observation.ell,
                    self.observation.c,
                    self.observation.o,
                )
            }
            evidence_by_component = {
                name: tuple(
                    item
                    for item in self.evidence
                    if item.robustness_component == name
                )
                for name in components
            }
            if any(not items for items in evidence_by_component.values()):
                raise ValueError(
                    "robustness availability requires evidence for every component",
                )
            for name, component in components.items():
                availability = {
                    item.availability for item in evidence_by_component[name]
                }
                if availability == {"OBSERVED"}:
                    if (
                        component.state not in {"ESTIMATED", "INCONCLUSIVE"}
                        or component.missingness.state != "COMPLETE"
                    ):
                        raise ValueError(
                            "all-observed robustness evidence requires "
                            "ESTIMATED or INCONCLUSIVE with COMPLETE missingness",
                        )
                elif availability == {"OBSERVED", "UNAVAILABLE"}:
                    if (
                        component.state != "INCONCLUSIVE"
                        or component.missingness.state != "PARTIAL"
                        or component.reason is None
                        or (component.value is None) != (component.interval is None)
                    ):
                        raise ValueError(
                            "mixed robustness evidence requires INCONCLUSIVE "
                            "with paired numeric fields if supplied, a reason, "
                            "and PARTIAL missingness",
                        )
                elif availability == {"UNAVAILABLE"}:
                    if (
                        component.state != "NOT_ESTIMABLE"
                        or component.value is not None
                        or component.interval is not None
                        or component.reason is None
                        or component.missingness.state != "UNAVAILABLE"
                    ):
                        raise ValueError(
                            "all-unavailable robustness evidence requires "
                            "NOT_ESTIMABLE with no value or interval, a reason, "
                            "and UNAVAILABLE missingness",
                        )
                else:
                    raise ValueError(
                        "robustness evidence has an unknown availability aggregate",
                    )
            parent_accounting = self.contract.cost_accounting
            component_accounting = self.observation.c.accounting
            if parent_accounting is None:
                raise ValueError("robustness contract requires parent accounting")
            if component_accounting != parent_accounting:
                raise ValueError(
                    "monetary-cost accounting must exactly match contract",
                )
            if self.observation.c.currency != self.contract.currency:
                raise ValueError("monetary-cost currency must match contract")
        elif self.contract.family == "OPERATIONAL_COST":
            if not isinstance(self.observation, OperationalCostObservation):
                raise ValueError(
                    "operational cost requires a typed nonnegative observation",
                )
            if self.observation.accounting != self.contract.cost_accounting:
                raise ValueError(
                    "operational-cost accounting must exactly match contract",
                )
            if self.observation.currency != self.contract.currency:
                raise ValueError("operational-cost currency must match contract")
        return self

    @computed_field
    @property
    def result_digest(self) -> str:
        payload = self.model_dump(mode="json", exclude={"result_digest"})
        payload["receipt_binding"] = receipt_binding(self)
        return canonical_sha256(payload)
